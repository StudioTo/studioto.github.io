function search(qRaw) {
    const q0 = qRaw.trim();
    const chars = [...(q0 || qRaw)];
    if (chars.length === 1) {
        const cp = chars[0].codePointAt(0);
        const hit = index.filter(x => x.it[0] === cp);
        if (hit.length) return hit;
    }
    const m = q0.match(/^(?:u\+?|0x)?([0-9a-f]{2,5})$/i);
    const codeHits = m ? index.filter(x => hex(x.it[0]) === m[1].toUpperCase().padStart(4, "0")) : [];
    const q = norm(q0);
    if (!q) return codeHits;
    const words = q.split(" ");
    const scored = [];
    for (const x of index) {
        if (codeHits.includes(x)) continue;
        if (!words.every(w => w.length <= 2 ? x.toks.has(w) : x.hay.includes(w))) continue;
        let s = 5;
        if (x.names.includes(q)) s = 0;
        else if (x.names.some(n => n.startsWith(q))) s = 1;
        else if (x.en === q || x.en.startsWith(q)) s = 2;
        else if (x.fr.includes(q)) s = 3;
        else if (x.en.includes(q)) s = 4;
        scored.push([s, x.en.length, x]);
    }
    scored.sort((a, b) => a[0] - b[0] || a[1] - b[1]);
    return codeHits.concat(scored.map(s => s[2]));
}
